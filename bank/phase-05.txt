=== DAY 76 | Create and Print a Linked List
--- statement
Build a singly linked list from `n` values (in order) and print it in the form `1 -> 2 -> 3 -> NULL`.

**Input:** `n`, then `n` integers.
**Output:** the list.
--- input
4
10 20 30 40
--- output
10 -> 20 -> 30 -> 40 -> NULL
--- complexity
Time: O(n) | Space: O(n)
--- code
#include <iostream>
using namespace std;

struct Node {
    int val;
    Node* next;
    Node(int v) : val(v), next(nullptr) {}
};

int main() {
    int n;
    cin >> n;
    Node* head = nullptr;
    Node* tail = nullptr;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        Node* node = new Node(x);
        if (!head) head = tail = node;
        else {
            tail->next = node;
            tail = node;
        }
    }
    for (Node* cur = head; cur; cur = cur->next) cout << cur->val << " -> ";
    cout << "NULL" << endl;
    return 0;
}

=== DAY 84 | Detect a Cycle (Floyd's Algorithm)
--- statement
Build a list of `n` values. If `pos >= 0`, connect the tail to the node at index `pos` (0-based) to form a cycle. Detect whether a cycle exists using slow/fast pointers.

**Input:** `n`, then `n` integers, then `pos` (-1 for no cycle).
**Output:** `true` or `false`.
--- input
4
3 2 0 -4
1
--- output
true
--- complexity
Time: O(n) | Space: O(1)
--- code
#include <iostream>
#include <vector>
using namespace std;

struct Node {
    int val;
    Node* next;
    Node(int v) : val(v), next(nullptr) {}
};

int main() {
    int n, pos;
    cin >> n;
    vector<Node*> nodes(n);
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        nodes[i] = new Node(x);
        if (i > 0) nodes[i - 1]->next = nodes[i];
    }
    cin >> pos;
    if (pos >= 0) nodes[n - 1]->next = nodes[pos];

    Node* slow = nodes[0];
    Node* fast = nodes[0];
    bool cycle = false;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) {
            cycle = true;
            break;
        }
    }
    cout << (cycle ? "true" : "false") << endl;
    return 0;
}

=== DAY 86 | Merge Two Sorted Linked Lists
--- statement
Merge two sorted linked lists into one sorted list by relinking nodes.

**Input:** `n`, then `n` sorted integers, then `m`, then `m` sorted integers.
**Output:** the merged list.
--- input
3
1 3 5
4
2 4 6 8
--- output
1 2 3 4 5 6 8
--- complexity
Time: O(n + m) | Space: O(1) extra
--- code
#include <iostream>
using namespace std;

struct Node {
    int val;
    Node* next;
    Node(int v) : val(v), next(nullptr) {}
};

Node* readList() {
    int n;
    cin >> n;
    Node dummy(0);
    Node* tail = &dummy;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        tail->next = new Node(x);
        tail = tail->next;
    }
    return dummy.next;
}

Node* mergeLists(Node* a, Node* b) {
    Node dummy(0);
    Node* tail = &dummy;
    while (a && b) {
        if (a->val <= b->val) { tail->next = a; a = a->next; }
        else { tail->next = b; b = b->next; }
        tail = tail->next;
    }
    tail->next = a ? a : b;
    return dummy.next;
}

int main() {
    Node* a = readList();
    Node* b = readList();
    Node* head = mergeLists(a, b);
    for (Node* cur = head; cur; cur = cur->next) cout << cur->val << (cur->next ? " " : "\n");
    return 0;

    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        Node* node = new Node(x);
        if (!head) head = tail = node;
        else {
            tail->next = node;
            node->prev = tail;
            tail = node;
        }
    }
    cin >> key;

    Node* cur = head;
    while (cur) {
        Node* nxt = cur->next;
        if (cur->val == key) {
            if (cur->prev) cur->prev->next = cur->next;
            else head = cur->next;
            if (cur->next) cur->next->prev = cur->prev;
            delete cur;
        }
        cur = nxt;
    }
    if (!head) cout << "EMPTY\n";
    for (Node* p = head; p; p = p->next) cout << p->val << (p->next ? " " : "\n");
    return 0;
}
