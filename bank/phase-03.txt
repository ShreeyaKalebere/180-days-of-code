=== DAY 37 | Binary Search
--- statement
Given a sorted array, find the index (0-based) of `target` using binary search. Print `-1` if not found.

**Input:** `n target`, then `n` sorted integers.
**Output:** the index.
--- input
7 23
2 5 8 12 16 23 38
--- output
5
--- complexity
Time: O(log n) | Space: O(1)
--- code
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, target;
    cin >> n >> target;
    vector<int> a(n);
    for (auto& x : a) cin >> x;

    int lo = 0, hi = n - 1, ans = -1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] == target) { ans = mid; break; }
        if (a[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    cout << ans << endl;
    return 0;
}

=== DAY 38 | First and Last Occurrence
--- statement
In a sorted array, find the first and last index of `target` using binary search. Print `-1 -1` if it is absent.

**Input:** `n target`, then `n` sorted integers.
**Output:** `first last`.
--- input
8 5
1 3 5 5 5 5 7 9
--- output
2 5
--- complexity
Time: O(log n) | Space: O(1)
--- code
#include <iostream>
#include <vector>
using namespace std;

int findBound(const vector<int>& a, int target, bool first) {
    int lo = 0, hi = (int)a.size() - 1, ans = -1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] == target) {
            ans = mid;
            if (first) hi = mid - 1;
            else lo = mid + 1;
        } else if (a[mid] < target) {
            lo = mid + 1;
        } else {
            hi = mid - 1;
        }
    }
    return ans;
}

int main() {
    int n, target;
    cin >> n >> target;
    vector<int> a(n);
    for (auto& x : a) cin >> x;
    cout << findBound(a, target, true) << " " << findBound(a, target, false) << endl;
    return 0;
}

=== DAY 40 | Bubble Sort
--- statement
Sort an array in ascending order using bubble sort. Stop early if a pass makes no swaps.

**Input:** `n`, then `n` integers.
**Output:** the sorted array.
--- input
6
5 1 4 2 8 0
--- output
0 1 2 4 5 8
--- complexity
Time: O(n^2) worst, O(n) best | Space: O(1)
--- code
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto& x : a) cin >> x;

    for (int i = 0; i < n - 1; i++) {
        bool swapped = false;
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j + 1]) {
                swap(a[j], a[j + 1]);
                swapped = true;
            }
        }
        if (!swapped) break;
    }
    for (int i = 0; i < n; i++) cout << a[i] << (i + 1 < n ? " " : "\n");
    return 0;
}

=== DAY 41 | Selection Sort

**Input:** `n`, then `n` integers.
**Output:** the sorted array.
--- input

--- complexity
Time: O(n^2) | Space: O(1)

    int n;

=== DAY 42 | Insertion Sort
--- statement
Sort an array in ascending order using insertion sort.

**Input:** `n`, then `n` integers.
**Output:** the sorted array.
--- input
6
12 11 13 5 6 1
--- output
1 5 6 11 12 13
--- complexity
Time: O(n^2) worst, O(n) best | Space: O(1)
--- code
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto& x : a) cin >> x;

    for (int i = 1; i < n; i++) {
        int key = a[i];
        int j = i - 1;
        while (j >= 0 && a[j] > key) {
            a[j + 1] = a[j];
            j--;
        }
        a[j + 1] = key;
    }
    for (int i = 0; i < n; i++) cout << a[i] << (i + 1 < n ? " " : "\n");
    return 0;
}

=== DAY 43 | Merge Sort
--- statement
Sort an array using merge sort (divide and conquer).

**Input:** `n`, then `n` integers.
**Output:** the sorted array.
--- input
7
38 27 43 3 9 82 10
--- output
3 9 10 27 38 43 82
--- complexity
Time: O(n log n) | Space: O(n)
--- code
#include <iostream>
#include <vector>
using namespace std;

void mergeHalves(vector<int>& a, int lo, int mid, int hi) {
    vector<int> tmp;
    tmp.reserve(hi - lo + 1);
    int i = lo, j = mid + 1;
    while (i <= mid && j <= hi) tmp.push_back(a[i] <= a[j] ? a[i++] : a[j++]);
    while (i <= mid) tmp.push_back(a[i++]);
    while (j <= hi) tmp.push_back(a[j++]);
    for (int k = 0; k < (int)tmp.size(); k++) a[lo + k] = tmp[k];
}

void mergeSort(vector<int>& a, int lo, int hi) {
    if (lo >= hi) return;
    int mid = lo + (hi - lo) / 2;
    mergeSort(a, lo, mid);
    mergeSort(a, mid + 1, hi);
    mergeHalves(a, lo, mid, hi);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto& x : a) cin >> x;
    mergeSort(a, 0, n - 1);
    for (int i = 0; i < n; i++) cout << a[i] << (i + 1 < n ? " " : "\n");
    return 0;
}

=== DAY 44 | Quick Sort
--- statement
Sort an array using quick sort with the Lomuto partition scheme (last element as pivot).

**Input:** `n`, then `n` integers.
**Output:** the sorted array.
--- input
6
10 7 8 9 1 5
--- output
1 5 7 8 9 10
--- complexity
Time: O(n log n) average, O(n^2) worst | Space: O(log n) stack
--- code
#include <iostream>
#include <vector>
using namespace std;

int partitionArr(vector<int>& a, int lo, int hi) {
    int pivot = a[hi];
    int i = lo - 1;
    for (int j = lo; j < hi; j++) {
        if (a[j] < pivot) swap(a[++i], a[j]);
    }
    swap(a[i + 1], a[hi]);
    return i + 1;
}

void quickSort(vector<int>& a, int lo, int hi) {
    if (lo >= hi) return;
    int p = partitionArr(a, lo, hi);
    quickSort(a, lo, p - 1);
    quickSort(a, p + 1, hi);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto& x : a) cin >> x;
    quickSort(a, 0, n - 1);
    for (int i = 0; i < n; i++) cout << a[i] << (i + 1 < n ? " " : "\n");
    return 0;
}

=== DAY 45 | Count Inversions
--- statement
An inversion is a pair `(i, j)` with `i < j` and `a[i] > a[j]`. Count inversions using a modified merge sort.

**Input:** `n`, then `n` integers.
**Output:** the number of inversions.
--- input
5
2 4 1 3 5
--- output
3
--- complexity
Time: O(n log n) | Space: O(n)
--- code
#include <iostream>
#include <vector>
using namespace std;

long long sortCount(vector<int>& a, int lo, int hi) {
    if (lo >= hi) return 0;
    int mid = lo + (hi - lo) / 2;
    long long count = sortCount(a, lo, mid) + sortCount(a, mid + 1, hi);
    vector<int> tmp;
    int i = lo, j = mid + 1;
    while (i <= mid && j <= hi) {
        if (a[i] <= a[j]) tmp.push_back(a[i++]);
        else {
            count += mid - i + 1;
            tmp.push_back(a[j++]);
        }
    }
    while (i <= mid) tmp.push_back(a[i++]);
    while (j <= hi) tmp.push_back(a[j++]);
    for (int k = 0; k < (int)tmp.size(); k++) a[lo + k] = tmp[k];
    return count;
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto& x : a) cin >> x;
    cout << sortCount(a, 0, n - 1) << endl;
    return 0;
}

=== DAY 46 | Search in a Rotated Sorted Array
--- statement

--- complexity

--- code

#include <vector>

=== DAY 51 | Container With Most Water
--- statement
Given heights of vertical lines, choose two lines that together with the x-axis hold the most water. Print the maximum area.

**Input:** `n`, then `n` heights.
**Output:** the maximum area.
--- input
9
1 8 6 2 5 4 8 3 7
--- output
49
--- complexity
Time: O(n) | Space: O(1)

#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> h(n);
    for (auto& x : h) cin >> x;

    }

=== DAY 53 | Sort Colors (Dutch National Flag)
--- statement
Sort an array containing only 0s, 1s and 2s in a single pass without using a sort function.

**Output:** the sorted array.
--- input
6

=== DAY 54 | Kth Smallest Element (Quickselect)
--- statement
Find the k-th smallest element (1-based) in an unsorted array using quickselect.

**Input:** `n k`, then `n` integers.
**Output:** the k-th smallest value.
--- input
6 3
7 10 4 3 20 15
--- output
7
--- complexity
Time: O(n) average | Space: O(1) extra
--- code
#include <iostream>
#include <vector>
using namespace std;

int partitionArr(vector<int>& a, int lo, int hi) {
    int pivot = a[hi], i = lo;
    for (int j = lo; j < hi; j++) {
        if (a[j] < pivot) swap(a[i++], a[j]);
    }
    swap(a[i], a[hi]);
    return i;
}

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (auto& x : a) cin >> x;

    int lo = 0, hi = n - 1, target = k - 1;
    while (lo <= hi) {
        int p = partitionArr(a, lo, hi);
        if (p == target) break;
        if (p < target) lo = p + 1;
        else hi = p - 1;
    }
    cout << a[target] << endl;
    return 0;
}
