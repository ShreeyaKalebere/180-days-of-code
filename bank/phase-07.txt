=== DAY 111 | Build a Binary Tree and Inorder Traversal
--- statement
Trees in this phase are given in **level order**, with `-1` meaning "no node". Build the tree and print its inorder traversal (left, root, right).

**Input:** `n`, then `n` level-order values.
**Output:** the inorder traversal.
--- input
7
1 2 3 4 5 -1 6
--- output
4 2 5 1 3 6
--- complexity
Time: O(n) | Space: O(n)
--- code
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

struct Node {
    int val;
    Node* left;
    Node* right;
    Node(int v) : val(v), left(nullptr), right(nullptr) {}
};

Node* buildLevelOrder(const vector<int>& a) {
    if (a.empty() || a[0] == -1) return nullptr;
    Node* root = new Node(a[0]);
    queue<Node*> q;
    q.push(root);
    size_t i = 1;
    while (!q.empty() && i < a.size()) {
        Node* cur = q.front();
        q.pop();
        if (i < a.size() && a[i] != -1) { cur->left = new Node(a[i]); q.push(cur->left); }
        i++;
        if (i < a.size() && a[i] != -1) { cur->right = new Node(a[i]); q.push(cur->right); }
        i++;
    }
    return root;
}

void inorder(Node* root, vector<int>& out) {
    if (!root) return;
    inorder(root->left, out);
    out.push_back(root->val);
    inorder(root->right, out);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto& x : a) cin >> x;
    Node* root = buildLevelOrder(a);
    vector<int> out;
    inorder(root, out);
    for (size_t i = 0; i < out.size(); i++) cout << out[i] << (i + 1 < out.size() ? " " : "\n");
    return 0;
}

**Output:** the diameter.

1 2 3 4 5

--- complexity
Time: O(n) | Space: O(h)

#include <vector>
#include <queue>
#include <algorithm>
using namespace std;

struct Node {
    int val;
    Node* left;
    Node* right;
    Node(int v) : val(v), left(nullptr), right(nullptr) {}
};

Node* buildLevelOrder(const vector<int>& a) {
    if (a.empty() || a[0] == -1) return nullptr;
    Node* root = new Node(a[0]);
    queue<Node*> q;
    q.push(root);

#include <vector>
#include <queue>
#include <algorithm>
#include <cstdlib>
using namespace std;

    int val;
    Node* left;
    Node* right;
    Node(int v) : val(v), left(nullptr), right(nullptr) {}
};

Node* buildLevelOrder(const vector<int>& a) {
    if (a.empty() || a[0] == -1) return nullptr;
    Node* root = new Node(a[0]);

        if (i < a.size() && a[i] != -1) { cur->left = new Node(a[i]); q.push(cur->left); }

=== DAY 121 | Lowest Common Ancestor in a Binary Tree

--- complexity

#include <vector>
#include <queue>
using namespace std;

struct Node {
    int val;
    Node* left;
    Node* right;
    Node(int v) : val(v), left(nullptr), right(nullptr) {}
};

Node* buildLevelOrder(const vector<int>& a) {
    if (a.empty() || a[0] == -1) return nullptr;
    Node* root = new Node(a[0]);
    queue<Node*> q;
    q.push(root);

        q.pop();
        if (i < a.size() && a[i] != -1) { cur->left = new Node(a[i]); q.push(cur->left); }

    if (!r || r->val == p || r->val == q) return r;

=== DAY 123 | Search in a BST
--- statement
Build a BST by inserting `n` values, then search for `key` iteratively. Print `Found` or `Not Found`.

**Input:** `n`, then `n` integers, then `key`.
**Output:** `Found` or `Not Found`.

8 3 10 1 6 14

--- output

--- complexity
Time: O(h) | Space: O(1) for the search
--- code
#include <iostream>
using namespace std;

struct Node {
    int val;
    Node* left;
    Node* right;
    Node(int v) : val(v), left(nullptr), right(nullptr) {}
};

    if (!r) return new Node(x);
    if (x < r->val) r->left = insert(r->left, x);
    else r->right = insert(r->right, x);
    return r;

        int x;

#include <climits>
using namespace std;

struct Node {
    int val;
    Node* left;
    Node* right;
    Node(int v) : val(v), left(nullptr), right(nullptr) {}
};

Node* buildLevelOrder(const vector<int>& a) {
    if (a.empty() || a[0] == -1) return nullptr;
    Node* root = new Node(a[0]);
    queue<Node*> q;
    q.push(root);
    size_t i = 1;
    while (!q.empty() && i < a.size()) {
        Node* cur = q.front();
        q.pop();
        if (i < a.size() && a[i] != -1) { cur->left = new Node(a[i]); q.push(cur->left); }
        i++;
        if (i < a.size() && a[i] != -1) { cur->right = new Node(a[i]); q.push(cur->right); }
        i++;
    }
    return root;
}

bool valid(Node* r, long long lo, long long hi) {
    if (!r) return true;
    if (r->val <= lo || r->val >= hi) return false;
    return valid(r->left, lo, r->val) && valid(r->right, r->val, hi);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto& x : a) cin >> x;
    cout << (valid(buildLevelOrder(a), LLONG_MIN, LLONG_MAX) ? "true" : "false") << endl;
    return 0;
}

=== DAY 125 | Minimum and Maximum in a BST
--- statement
Build a BST from `n` inserted values and print its minimum and maximum by walking left and right.

**Input:** `n`, then `n` integers.
**Output:** `min max`.
--- input
6
15 10 20 8 12 25
--- output
8 25
--- complexity
Time: O(h) | Space: O(1) for the walk
--- code
#include <iostream>
using namespace std;

struct Node {
    int val;
    Node* left;
    Node* right;
    Node(int v) : val(v), left(nullptr), right(nullptr) {}
};

Node* insert(Node* r, int x) {
    if (!r) return new Node(x);
    if (x < r->val) r->left = insert(r->left, x);
    else r->right = insert(r->right, x);
    return r;
}

int main() {
    int n;
    cin >> n;
    Node* root = nullptr;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        root = insert(root, x);
    }
    Node* lo = root;
    while (lo->left) lo = lo->left;
    Node* hi = root;
    while (hi->right) hi = hi->right;
    cout << lo->val << " " << hi->val << endl;
    return 0;
}

=== DAY 126 | Kth Smallest Element in a BST
--- statement
Build a BST from `n` inserted values and find the k-th smallest value using an inorder traversal that stops early.

**Input:** `n k`, then `n` integers.
**Output:** the k-th smallest value.
--- input
6 3
5 3 6 2 4 1
--- output
3
--- complexity
Time: O(h + k) | Space: O(h)
--- code
#include <iostream>
using namespace std;

struct Node {
    int val;
    Node* left;
    Node* right;
    Node(int v) : val(v), left(nullptr), right(nullptr) {}
};

Node* insert(Node* r, int x) {
    if (!r) return new Node(x);
    if (x < r->val) r->left = insert(r->left, x);
    else r->right = insert(r->right, x);
    return r;
}

int counter = 0, answer = -1;

void inorder(Node* r, int k) {
    if (!r || answer != -1) return;
    inorder(r->left, k);
    if (++counter == k) {
        answer = r->val;
        return;
    }
    inorder(r->right, k);
}

int main() {
    int n, k;
    cin >> n >> k;
    Node* root = nullptr;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        root = insert(root, x);
    }
    inorder(root, k);
    cout << answer << endl;
    return 0;
}

=== DAY 127 | Delete a Node from a BST
--- statement
Build a BST from `n` inserted values, delete `key` (handle 0, 1 and 2 children; use the inorder successor), and print the inorder traversal.

**Input:** `n`, then `n` integers, then `key`.
**Output:** the inorder traversal after deletion.
--- input
7
50 30 70 20 40 60 80
50
--- output
20 30 40 60 70 80
--- complexity
Time: O(h) | Space: O(h)
--- code
#include <iostream>
#include <vector>
using namespace std;

struct Node {
    int val;
    Node* left;
    Node* right;
    Node(int v) : val(v), left(nullptr), right(nullptr) {}
};

Node* insert(Node* r, int x) {
    if (!r) return new Node(x);
    if (x < r->val) r->left = insert(r->left, x);
    else r->right = insert(r->right, x);
    return r;
}

Node* removeKey(Node* r, int key) {
    if (!r) return nullptr;
    if (key < r->val) r->left = removeKey(r->left, key);
    else if (key > r->val) r->right = removeKey(r->right, key);
    else {
        if (!r->left) { Node* t = r->right; delete r; return t; }
        if (!r->right) { Node* t = r->left; delete r; return t; }
        Node* succ = r->right;
        while (succ->left) succ = succ->left;
        r->val = succ->val;
        r->right = removeKey(r->right, succ->val);
    }
    return r;
}

void inorder(Node* r, vector<int>& out) {
    if (!r) return;
    inorder(r->left, out);
    out.push_back(r->val);
    inorder(r->right, out);
}

int main() {
    int n, key;
    cin >> n;
    Node* root = nullptr;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        root = insert(root, x);
    }
    cin >> key;
    root = removeKey(root, key);
    vector<int> out;
    inorder(root, out);
    for (size_t i = 0; i < out.size(); i++) cout << out[i] << (i + 1 < out.size() ? " " : "\n");
    return 0;
}

=== DAY 128 | Kth Largest Element Using a Min-Heap
--- statement
Find the k-th largest element by keeping a min-heap (`priority_queue`) of size `k`.

**Input:** `n k`, then `n` integers.
**Output:** the k-th largest value.
--- input
6 2
3 2 1 5 6 4
--- output
5
--- complexity
Time: O(n log k) | Space: O(k)
--- code
#include <iostream>
#include <queue>
#include <vector>
#include <functional>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    priority_queue<int, vector<int>, greater<int>> heap;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        heap.push(x);
        if ((int)heap.size() > k) heap.pop();
    }
    cout << heap.top() << endl;
    return 0;
}

=== DAY 129 | Heap Sort
--- statement
Sort an array in ascending order using heap sort (build a max-heap, then repeatedly move the max to the end).

**Input:** `n`, then `n` integers.
**Output:** the sorted array.
--- input
6
12 11 13 5 6 7
--- output
5 6 7 11 12 13
--- complexity
Time: O(n log n) | Space: O(1)
--- code
#include <iostream>
#include <vector>
using namespace std;

void heapify(vector<int>& a, int n, int i) {
    while (true) {
        int largest = i, l = 2 * i + 1, r = 2 * i + 2;
        if (l < n && a[l] > a[largest]) largest = l;
        if (r < n && a[r] > a[largest]) largest = r;
        if (largest == i) return;
        swap(a[i], a[largest]);
        i = largest;
    }
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto& x : a) cin >> x;
    for (int i = n / 2 - 1; i >= 0; i--) heapify(a, n, i);
    for (int end = n - 1; end > 0; end--) {
        swap(a[0], a[end]);
        heapify(a, end, 0);
    }
    for (int i = 0; i < n; i++) cout << a[i] << (i + 1 < n ? " " : "\n");
    return 0;
}

=== DAY 130 | Merge K Sorted Arrays
--- statement
Merge `k` sorted arrays into one sorted array using a min-heap.

**Input:** `k`, then for each array its length followed by its elements.
**Output:** the merged array.
--- input
3
3 1 4 7
2 2 5
4 0 3 6 9
--- output
0 1 2 3 4 5 6 7 9
--- complexity
Time: O(N log k) | Space: O(k)
--- code
#include <iostream>
#include <vector>
#include <queue>
#include <tuple>
#include <functional>
using namespace std;

int main() {
    int k;
    cin >> k;
    vector<vector<int>> arrs(k);
    for (auto& arr : arrs) {
        int len;
        cin >> len;
        arr.resize(len);
        for (auto& x : arr) cin >> x;
    }
    using Item = tuple<int, int, int>;  // value, array index, element index
    priority_queue<Item, vector<Item>, greater<Item>> heap;
    for (int i = 0; i < k; i++)
        if (!arrs[i].empty()) heap.push({arrs[i][0], i, 0});

    vector<int> out;
    while (!heap.empty()) {
        auto [val, ai, ei] = heap.top();
        heap.pop();
        out.push_back(val);
        if (ei + 1 < (int)arrs[ai].size()) heap.push({arrs[ai][ei + 1], ai, ei + 1});
    }
    for (size_t i = 0; i < out.size(); i++) cout << out[i] << (i + 1 < out.size() ? " " : "\n");
    return 0;
}
